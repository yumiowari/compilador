É a gramática:

G = (V, Σ, P, PGRM).

V = {PGRM, DECL, CMD, TIPO, SEQ_ID, TIPO_ARR, SEQ_ARR, EXPR_REL,
     CMD_REPET, EXPR, CMD_PRINT, ESCREVA_REC, CMD_SCANF, CMD_ATRIB, TERMO, FATOR}.

Σ = {PROGRAMA, ID, INICIO, FIM, ARR, INTEIRO, REAL, CARACTER, ',', CADEIA, LISTA_INT,
     LISTA_REAL, '[', NUM, ']', ENQUANTO, FIM_ENQUANTO, SE, ENTAO, FIM_SE, .M., .m.,
     .I., ESCREVA, STR, LEIA, :=, '+', '-', '*', '/', '(', ')'}.

P = 
{
    PGRM -> PROGRAMA ID INICIO DECL CMD FIM;

    DECL -> TIPO ID SEQ_ID DECL       |
            TIPO_ARR ARR SEQ_ARR DECL |
            ε                         ;

    TIPO -> INTEIRO  |
            REAL     |
            CARACTER ;

    SEQ_ID -> ',' ID SEQ_ID |
              ε             ;

    TIPO_ARR -> CADEIA     |
                LISTA_INT  |
                LISTA_REAL ;

    ARR -> ID '[' NUM ']' |
           ID '[' ID ']'  ;

    SEQ_ARR -> ',' ARR SEQ_ARR |
               ε               ;

    CMD -> CMD_REPET CMD |
           CMD_PRINT CMD |
           CMD_SCANF CMD |
           CMD_ATRIB CMD |
           ε             ;

    CMD_REPET -> ENQUANTO EXPR_REL CMD FIM_ENQUANTO CMD |
                 SE EXPR_REL ENTAO CMD FIM_SE CMD       ;

    EXPR_REL -> EXPR .M. EXPR |
                EXPR .m. EXPR |
                EXPR .I. EXPR ;

    CMD_PRINT -> ESCREVA STR ESCREVA_REC |
                 ESCREVA ID ESCREVA_REC  |
                 ESCREVA ARR ESCREVA_REC ;

    ESCREVA_REC -> ',' STR ESCREVA_REC |
                   ',' ID ESCREVA_REC  |
                   ',' ARR ESCREVA_REC |
                   ε                   ;

    SEQ_STR -> ',' STR SEQ_STR |
               ε               ;

    CMD_SCANF -> LEIA ID SEQ_ID   |
                 LEIA ARR SEQ_ARR ;

    CMD_ATRIB -> ID SEQ_ID := EXPR CMD;

    EXPR -> TERMO          |
            EXPR '+' TERMO |
            EXPR '-' TERMO ;

    TERMO -> FATOR           |
             TERMO '*' FATOR |
             TERMO '/' FATOR ;

    FATOR -> '(' EXPR ')' |
             NUM          |
             ID           |
             ARR          ;
}
